### 自定义新建表格 + 写入/读取 区域内容 + 统计
```js
const dateText = await Input.textAsync("Input the date info (e.g. 2026-07-07)");
const sheet = Workbook.insertSheet();
sheet.setName("newSheet");
Output.log(`new sheet created: ${sheet.getName()}`);

sheet.getRange("A1:D1").setValues([
    ["Date", "Name", "Content", "Status"]
]);
sheet.getRange("A2:D6").setValues([
    [dateText, "Tom", "Customer Experience", "Completed"],
    [dateText, "Ben", "Data Extract", "WIP"],
    [dateText, "Steven", "Strategy", "Not Started"],
    [dateText, "Marry", "Operation", "Not Started"],
    [dateText, "Nick", "Finance", "Not Started"]
]);
Output.log("Update Done !");

const values = sheet.getRange("A2:D6").getDisplayValues();
let doneCount = 0;
let doingCount = 0;
let todoCount = 0;
for (let i=0; i<values.length; i++) {
    const status = values[i][3];
    if (status === "Completed") {
        doneCount++;
    } else if (status === "WIP") {
        doingCount++;
    } else if (status === "Not Started") {
        todoCount++;
    } 
}

sheet.getRange("E1:F4").setValues([
  ["状态", "数量"],
  ["完成", doneCount],
  ["处理中", doingCount],
  ["未开始", todoCount]
]);
Output.log("统计完成");
```

### 过滤数据
```js
const values = sheet.getRange("A1:D100").getValues();
const header = values[0];
const rows = values.slice(1);

const completedRows = rows.filter(row => row[3] === "Completed");
const output = [
    header,
    ...completedRows
];

sheet.getRange(`F1:I${output.length}`).setValues(output);
```

### 多列批量计算
```js
const values = sheet.getRange("A2:C6").getValues();
const results = [];

for (let i = 0; i < values.length; i++) {
    const price = Number(values[i][0]);
    const qty = Number(values[i][1]);
    const discount = Number(values[i][2]);

    const amount = price * qty;
    const finalAmount = amount * discount;
    results.push([amount, finalAmount]);
}

sheet.getRange("D2:E6").setValues(results);
```

### 清洗空行

function isEmptyRow(row) {
    return row.every(cell => cell === "" || cell === null || cell === undefined);
}

function formatDateTime(date) {
    const y = date.getFullYear();
    const m = String(date.getMonth() + 1).padStart(2, "0");
    const d = String(date.getDate()).padStart(2, "0");

    const hh = String(date.getHours()).padStart(2, "0");
    const mm = String(date.getMinutes()).padStart(2, "0");
    const ss = String(date.getSeconds()).padStart(2, "0");

    return `${y}-${m}-${d} ${hh}:${mm}:${ss}`;
}

function addDays(date, days) {
    const result = new Date(date);
    result.setDate(result.getDate() + days);
    return result;
}

function daysBetween(date1, date2) {
    const d1 = normalizeDate(date1);
    const d2 = normalizeDate(date2);

    const diffMs = d2.getTime() - d1.getTime();  // 日期时间转换成毫秒时间戳

    return Math.floor(diffMs / (1000 * 60 * 60 * 24));
}

function toDate(value) {
    if (value instanceof Date) {
        return value;
    }

    if (!value) {
        return null;
    }

    const date = new Date(value);

    if (isNaN(date.getTime())) {
        return null;
    }

    return date;
}

function isWeekend(date) {
    const day = date.getDay();

    return day === 0 || day === 6;
}

function formatInTimeZone(date, timeZone) {
    const parts = new Intl.DateTimeFormat("en-CA", {
        timeZone: timeZone,
        year: "numeric",
        month: "2-digit",
        day: "2-digit",
        hour: "2-digit",
        minute: "2-digit",
        second: "2-digit",
        hour12: false
    }).formatToParts(date);

    const map = {};

    for (let i = 0; i < parts.length; i++) {
        map[parts[i].type] = parts[i].value;
    }

    return `${map.year}-${map.month}-${map.day} ${map.hour}:${map.minute}:${map.second}`;
}

function formatWithOffset(date, offsetHours) {
    const utcMs = date.getTime() + date.getTimezoneOffset() * 60 * 1000;
    const targetDate = new Date(utcMs + offsetHours * 60 * 60 * 1000);

    return formatDateTimeLocal(targetDate);
}
function formatDateTimeLocal(date) {
    const y = date.getFullYear();

}
