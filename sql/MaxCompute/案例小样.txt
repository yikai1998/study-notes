-- 创建一个假期/上班/调休的日历表
create table if not exists cyk_dim_calendar (
    cal_date      datetime comment "日期(当天00:00:00)",
    is_workday_cn bigint comment "中国大陆工作日 1=是 0=否",
    is_workday_hk bigint comment "中国香港工作日 1=是 0=否",
    is_workday_us bigint comment "美国工作日(联邦假日) 1=是 0=否"
) comment "SLA工作日历";

insert overwrite table cyk_dim_calendar
select  cal_date
       ,case  -- 中国大陆: 调休上班>法定假>默认双休
          when cal_date in (
               to_date("2026-01-04", "yyyy-mm-dd")  -- 元旦调休 周日
              ,to_date("2026-02-14", "yyyy-mm-dd")  -- 春节调休 周六
              ,to_date("2026-02-28", "yyyy-mm-dd")  -- 春节调休 周六
              ,to_date("2026-05-09", "yyyy-mm-dd")  -- 劳动节调休 周六
              ,to_date("2026-09-20", "yyyy-mm-dd")  -- 国庆调休 周日
              ,to_date("2026-10-10", "yyyy-mm-dd")  -- 国庆调休 周六
               ) then 1
          when cal_date in (
               to_date("2026-01-01", "yyyy-mm-dd"), to_date("2026-01-02", "yyyy-mm-dd"), to_date("2026-01-03", "yyyy-mm-dd")  -- 元旦
              ,to_date("2026-02-15", "yyyy-mm-dd"), to_date("2026-02-16", "yyyy-mm-dd"), to_date("2026-02-17", "yyyy-mm-dd")  -- 春节 2/15-2/23
              ,to_date("2026-02-18", "yyyy-mm-dd"), to_date("2026-02-19", "yyyy-mm-dd"), to_date("2026-02-20", "yyyy-mm-dd")
              ,to_date("2026-02-21", "yyyy-mm-dd"), to_date("2026-02-22", "yyyy-mm-dd"), to_date("2026-02-23", "yyyy-mm-dd")
              ,to_date("2026-04-04", "yyyy-mm-dd"), to_date("2026-04-05", "yyyy-mm-dd"), to_date("2026-04-06", "yyyy-mm-dd")  -- 清明
              ,to_date("2026-05-01", "yyyy-mm-dd"), to_date("2026-05-02", "yyyy-mm-dd"), to_date("2026-05-03", "yyyy-mm-dd")  -- 劳动节 5/1-5/5
              ,to_date("2026-05-04", "yyyy-mm-dd"), to_date("2026-05-05", "yyyy-mm-dd")
              ,to_date("2026-06-19", "yyyy-mm-dd"), to_date("2026-06-20", "yyyy-mm-dd"), to_date("2026-06-21", "yyyy-mm-dd")  -- 端午
              ,to_date("2026-09-25", "yyyy-mm-dd"), to_date("2026-09-26", "yyyy-mm-dd"), to_date("2026-09-27", "yyyy-mm-dd")  -- 中秋
              ,to_date("2026-10-01", "yyyy-mm-dd"), to_date("2026-10-02", "yyyy-mm-dd"), to_date("2026-10-03", "yyyy-mm-dd")  -- 国庆 10/1-10/7
              ,to_date("2026-10-04", "yyyy-mm-dd"), to_date("2026-10-05", "yyyy-mm-dd"), to_date("2026-10-06", "yyyy-mm-dd")
              ,to_date("2026-10-07", "yyyy-mm-dd")
               ) then 0
          when weekday(cal_date) >= 5 then 0
          else 1
        end as is_workday_cn
       ,case  -- 中国香港: 无调休, 周末+公众假期(含补假/增补假)
          when cal_date in (
               to_date("2026-01-01", "yyyy-mm-dd")  -- 一月一日
              ,to_date("2026-02-17", "yyyy-mm-dd"), to_date("2026-02-18", "yyyy-mm-dd"), to_date("2026-02-19", "yyyy-mm-dd")  -- 农历年初一至三
              ,to_date("2026-04-03", "yyyy-mm-dd"), to_date("2026-04-04", "yyyy-mm-dd")  -- 耶稣受难节及翌日
              ,to_date("2026-04-06", "yyyy-mm-dd"), to_date("2026-04-07", "yyyy-mm-dd")  -- 清明节翌日, 复活节星期一翌日(增补)
              ,to_date("2026-05-01", "yyyy-mm-dd")  -- 劳动节
              ,to_date("2026-05-25", "yyyy-mm-dd")  -- 佛诞翌日
              ,to_date("2026-06-19", "yyyy-mm-dd")  -- 端午节
              ,to_date("2026-07-01", "yyyy-mm-dd")  -- 特区成立纪念日
              ,to_date("2026-09-26", "yyyy-mm-dd")  -- 中秋节翌日
              ,to_date("2026-10-01", "yyyy-mm-dd")  -- 国庆日
              ,to_date("2026-10-19", "yyyy-mm-dd")  -- 重阳节翌日
              ,to_date("2026-12-25", "yyyy-mm-dd"), to_date("2026-12-26", "yyyy-mm-dd")  -- 圣诞节及圣诞后第一个周日
               ) then 0
          when weekday(cal_date) >= 5 then 0  -- 周六日休
          else 1
        end as is_workday_hk
       ,case  -- 美国: 无调休, 周末+联邦假日(逢周六提前周五, 逢周日顺延周一)
          when cal_date in (
               to_date("2026-01-01", "yyyy-mm-dd")  -- new year's day
              ,to_date("2026-01-19", "yyyy-mm-dd")  -- martin luther king jr. day
              ,to_date("2026-02-16", "yyyy-mm-dd")  -- washington's birthday
              ,to_date("2026-05-25", "yyyy-mm-dd")  -- memorial day
              ,to_date("2026-06-19", "yyyy-mm-dd")  -- juneteenth
              ,to_date("2026-07-03", "yyyy-mm-dd")  -- independence day (7/4逢周六, 观察日周五)
              ,to_date("2026-09-07", "yyyy-mm-dd")  -- labor day
              ,to_date("2026-10-12", "yyyy-mm-dd")  -- columbus day
              ,to_date("2026-11-11", "yyyy-mm-dd")  -- veterans day
              ,to_date("2026-11-26", "yyyy-mm-dd")  -- thanksgiving day
              ,to_date("2026-12-25", "yyyy-mm-dd")  -- christmas day
               ) then 0
          when weekday(cal_date) >= 5 then 0
          else 1
        end as is_workday_us
from (
    select dateadd(to_date("2026-01-01", "yyyy-mm-dd"), t.pos, "dd") as cal_date
    from (
        select posexplode(split(repeat("x,", 364), ",")) as (pos, val)
    ) t
    where t.pos <= 364
) d;
