LENGTH / CHAR_LENGTH
```sql
-- LENGTH 返回的是字节数，不是字符数！
SELECT LENGTH('abc');        -- 3
SELECT LENGTH('你好');        -- 6 ← 不是 2！UTF-8 中文占 3 字节

-- 如果要字符数用 CHAR_LENGTH
SELECT CHAR_LENGTH('你好');   -- 2
```

TRIM
```sql
-- 去掉首尾空格，数据清洗很常用
SELECT TRIM('  airwallex  ');           -- 'airwallex'
SELECT TRIM(LEADING  '-' FROM '--abc'); -- 'abc'
SELECT TRIM(TRAILING '-' FROM 'abc--'); -- 'abc'
```

查找位置
```sql
SELECT
INSTR('airwallexair', '@') AS at_pos,  -- 0
LOCATE('@', 'airwalledddx@air') AS at_pos_2  -- 13
```

SUBSTRING_INDEX  -- 按分隔符截取字符串语法返回一个str 的子字符串
```sql
SELECT
SUBSTRING_INDEX('aq@ws@asd', '@', 1) AS a,  -- aq
SUBSTRING_INDEX('aq@ws@asd', '@', 2) AS b,  -- aq@ws
SUBSTRING_INDEX('aq@ws@asd', '@', 3) AS c,  -- aq@ws@asd
SUBSTRING_INDEX('aq@ws@asd', '@', -1) AS d,  -- asd
SUBSTRING_INDEX('aq@ws@asd', '@', -2) AS e  -- ws@asd
```

REGEXP_LIKE
REGEXP_REPLACE
REGEXP_SUBSTR
```sql
-- REGEXP_LIKE：是否匹配
SELECT REGEXP_LIKE('airwallex@gmail.com', '@');  -- 1

-- REGEXP_REPLACE：替换匹配到的内容
SELECT REGEXP_REPLACE('abc123def', '[0-9]+', '#');  -- 'abc#def'

-- REGEXP_SUBSTR：提取匹配到的内容
SELECT REGEXP_SUBSTR('airwallex@gmail.com', '[a-z]+@[a-z]+');  -- 'airwallex@gmail'
```
