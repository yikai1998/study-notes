import os

## 文件的读写
# mac: "/";  window: "\\"; ~ 是“家目录”的简写，expanduser() 是把这个简写展开成真实路径。
# 不同操作系统的路径分隔符可能不同，建议使用 os.path.join() 来拼接路径
desktop = os.path.join(os.path.expanduser("~"), "Desktop")
print(desktop)

# 打开文件并读取内容 [the file must exist already]
file_path = os.path.join(desktop, '三字经.txt')
with open(file=file_path, mode='r', encoding='utf-8') as f:
    content = f.read()  # read() 适合小文件（十几MB以下都没问题）；如果文件很大，不建议一次性全部读入内存。
print(content)

# 整体覆盖 [if the file exists, then replace; if not, then would create this file]
with open(file=os.path.join(desktop, '三字经2.txt'), mode='w', encoding='utf-8') as f:
    f.write('[如果你看到了这行文字，说明之前的内容全部被我覆盖了]')

# 追加内容在末尾 [if the file does not exist, then it would create the file]
with open(file=os.path.join(desktop, '三字经3.txt'), mode='a', encoding='utf-8') as f:
    f.write('这是一行新的内容！\n')

# 读取文件的特定某行，如第3行
with open(file=os.path.join(desktop, '三字经.txt'), mode='r', encoding='utf-8') as f:
    for i, line in enumerate(f, start=1):
        if i == 3:
            print(line)
            break
    或：
    lines = f.readlines()
    if len(lines) >= 3:
        print(lines[2])
    else:
        print('文件不足3行')

"""
read 一次性读取整个文件的内容，将其作为一个字符串返回。
每次调用 readline 方法时，它会从文件的当前指针位置开始读取一行，并返回该行的字符串。行的末尾通常包括一个换行符 \n，除非在最后一行没有换行符。
连续调用 readline() 会接着往下读，不会每次都从第一行开始。
readlines 读取文件中的所有行，并将它们作为一个列表返回，其中每一行是一个列表元素，行末包含换行符。
read 适合读取较小的文件或需要处理整个文件内容为一个字符串的情况，
readlines() 适合需要一次性拿到所有行列表的情况，
“逐行处理”，其实更推荐直接： `for line in f:`
如果文件非常大，考虑使用文件对象的迭代操作来逐行读取文件，以减少内存使用。
"""
